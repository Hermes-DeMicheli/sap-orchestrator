  key client_id,

Exercise 5 -- AMDP Method

Write an AMDP method that calculates the total revenue per driver for a given date range.

Task

Use SQLScript with parameters and return a table with driver_id and total_revenue.

CLASS zcl_orch_amdp DEFINITION
  PUBLIC
  FINAL
  CREATE PUBLIC.
  PUBLIC SECTION.
    INTERFACES if_amdp_marker_hdb.
    CLASS-METHODS get_driver_revenue
      IMPORTING iv_date_from TYPE d
                iv_date_to   TYPE d
      RETURNING VALUE(rt_result) TYPE TABLE OF zdriver_revenue.
ENDCLASS.

CLASS zcl_orch_amdp IMPLEMENTATION.
  METHOD get_driver_revenue BY DATABASE FUNCTION
    FOR HDB
    LANGUAGE SQLSCRIPT
    OPTIONS READ-ONLY
    USING ztravel.
    rt_result = SELECT driver_id, SUM( distance_km * price_per_km ) AS total_revenue
                FROM ztravel
                WHERE travel_date BETWEEN :iv_date_from AND :iv_date_to
                GROUP BY driver_id;
  ENDMETHOD.
ENDCLASS.

Exercise 7 -- Clean ABAP: Value Object

Create an immutable value object for TravelData with validation in the constructor.

Task

Define a class with FINAL, no public setter, and a factory method that validates input.

CLASS zcl_travel_data DEFINITION
  FINAL
  CREATE PRIVATE.
  PUBLIC SECTION.
    DATA: mv_travel_id TYPE string READ-ONLY,
          mv_driver_id TYPE string READ-ONLY,
          mv_distance  TYPE i READ-ONLY.
    CLASS-METHODS create
      IMPORTING iv_travel_id TYPE string
                iv_driver_id TYPE string
                iv_distance  TYPE i
      RETURNING VALUE(ro_instance) TYPE REF TO zcl_travel_data.
  PRIVATE SECTION.
    METHODS constructor
      IMPORTING iv_travel_id TYPE string
                iv_driver_id TYPE string
                iv_distance  TYPE i.
ENDCLASS.

CLASS zcl_travel_data IMPLEMENTATION.
  METHOD create.
    IF iv_distance <= 0.
      RAISE EXCEPTION TYPE zcx_orch_error
        EXPORTING iv_message = 'Distance must be positive'
                  iv_code    = 'INVALID_DISTANCE'.
    ENDIF.
    ro_instance = NEW #( iv_travel_id = iv_travel_id
                         iv_driver_id = iv_driver_id
                         iv_distance  = iv_distance ).
  ENDMETHOD.
  METHOD constructor.
    mv_travel_id = iv_travel_id.
    mv_driver_id = iv_driver_id.
    mv_distance  = iv_distance.
  ENDMETHOD.
ENDCLASS.
